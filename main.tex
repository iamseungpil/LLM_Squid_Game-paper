%% LLM Squid Game --- KDD 2026 Undergraduate Consortium submission
%% Shared LaTeX skeleton for English (en/) and Korean (ko/) versions.
%% Compile with XeLaTeX:  xelatex main.tex && bibtex main && xelatex main.tex && xelatex main.tex
%% Switch language by changing the value of \paperLang below.

%% Load mathtools BEFORE acmart's class-time unicode-math autoload to silence
%% "Package mathtools Warning: For best results, please load mathtools *before*
%% unicode-math" under XeLaTeX/LuaLaTeX.
\RequirePackage{mathtools}

%% Known cosmetic warnings that we accept rather than silence:
%%   - "Package fontspec Warning: OpenType feature 'Numbers=Uppercase' (lnum)
%%     not available for font 'LinBiolinum_RB' / 'LinLibertine_RZI'." This
%%     comes from acmart's Linux Libertine font setup; the locally installed
%%     font files lack the lnum/tnum OpenType features but the rendered
%%     output is correct (digits fall back to the default numeral set).
%%     fontspec emits these via a non-standard warning macro that ignores
%%     the silence package, so we leave them as-is rather than patch
%%     fontspec internals.
%%   - "Package unicode-math Warning: \overbracket / \underbracket
%%     overwritten." A known interaction between unicode-math and amsmath
%%     under XeLaTeX. Output is unaffected.

\documentclass[sigconf, nonacm]{acmart}

%% ---------------------------------------------------------------------------
%% Language switch
%%   en  : English (original KDD-UC submission, body in v4-temp/main.tex)
%%   ko  : Korean translation (body retranslated 2026-05-08, prose only)
%% ---------------------------------------------------------------------------
\newcommand{\paperLang}{ko}  % en | ko

\AtBeginDocument{%
  \providecommand\BibTeX{{%
    Bib\TeX}}}

%% Citation style: acmart loads natbib automatically and, for sigconf,
%% defaults to acmnumeric (numbered citations like [1]) per the ACM
%% Reference Format. We DO NOT override with \citestyle{acmauthoryear}
%% because KDD/ACM convention is numeric. With this default:
%%   \citet{key} renders as "Author [N]"  (author appears in text flow)
%%   \citep{key} renders as "[N]"          (purely parenthetical)
%%   \cite{key}  renders as "[N]"          (equivalent to \citep here)
%% The bibliography list is auto-generated by ACM-Reference-Format.bst in
%% the order of first citation in the body.

%% Rights management --- KDD-UC is a non-archival venue, so we keep nonacm and
%% suppress the ACM Reference Format on the title page. The conference info
%% is preserved per the CFP.
\settopmatter{printacmref=false}
\setcopyright{none}
\copyrightyear{2026}
\acmYear{2026}
\acmConference[KDD-UC '26]{KDD 2026 Undergraduate Consortium}{August 9--13, 2026}{Jeju, Republic of Korea}
\acmISBN{}

\usepackage{booktabs}
\usepackage{graphicx}
\usepackage{array}
\usepackage{multirow}
\usepackage{amsmath}
\let\Bbbk\relax
\usepackage{amssymb}
\usepackage{xcolor}
\usepackage{listings}
\lstset{basicstyle=\ttfamily\small, breaklines=true, columns=fullflexible}
\usepackage{longtable}
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

%% Korean glyph rendering --- only loaded when \paperLang is "ko".
%% kotex requires XeLaTeX (or LuaLaTeX); pdflatex compilation is unsupported.
\usepackage{ifthen}
\ifthenelse{\equal{\paperLang}{ko}}{\usepackage{kotex}}{}

%% Shared figure path --- both languages reuse the same images.
\graphicspath{{figures/}}

%% Line-breaking safety net for narrow ACM columns. Long typewriter terms with
%% underscores lack natural break points, so we widen the emergency stretch
%% and tolerance to absorb the rest.
\setlength{\emergencystretch}{3em}
\tolerance=2000
\hbadness=10000

%% Tables numbered as Table X.Y where X is the parent section number.
\numberwithin{table}{section}

%% Bibliography compactness --- KDD-UC '26 allots 1 page for references, so we
%% trim natbib's inter-entry spacing and shrink the bibliography font one
%% step. Output remains within the ACM Reference Format style.
\setlength{\bibsep}{0pt plus 0.3ex}
\renewcommand{\bibfont}{\footnotesize}

\begin{document}

\input{\paperLang/content}

\end{document}
