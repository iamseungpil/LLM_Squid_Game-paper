%% Korean glyph rendering --- only loaded when \paperLang is "ko".
%% kotex requires XeLaTeX (or LuaLaTeX); mirrors the ko branch of
%% main_kdduc_acm.tex. Korean fonts ship without italic shapes, so the
%% "Font shape ... undefined" substitution warnings are cosmetic.
\usepackage{ifthen}
\ifthenelse{\equal{\paperLang}{ko}}{\usepackage{kotex}}{}

%% Shared figure path --- both languages reuse the same images.
\graphicspath{{figures/}}

%% Draft markers. \todo prints in red so that every unfilled number, missing
%% citation and unverified claim is visible in the PDF. Remove before
%% camera-ready (grep for \todo).
\newcommand{\todo}[1]{\textcolor{red}{[TODO: #1]}}

%% Accessibility alt-text hook. The ACM class defines \Description for
%% screen-reader text on figures; this article class does not, so the v1
%% appendix figure (ported from the ACM build) needs a local stand-in. Kept
%% as a silent no-op here since ICLR has no equivalent rendering path for it.
