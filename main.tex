%% LLM Squid Game --- ICLR 2026 submission ("Pricing Self-Preservation in Score Units")
%% Shared LaTeX skeleton for English (en/) and Korean (ko/) versions.
%% The earlier KDD-UC (ACM sigconf) entry point is kept as main_kdduc_acm.tex.
%%
%% Compile (XeLaTeX, both languages):
%%   xelatex main.tex && bibtex main && xelatex main.tex && xelatex main.tex
%% pdflatex also works for the English build; the Korean build needs XeLaTeX
%% because kotex is loaded only under \paperLang = ko.
%% Switch language by changing the value of \paperLang below.

\documentclass{article}
\usepackage{iclr2026_conference,times}

%% ---------------------------------------------------------------------------
%% Language switch
%%   en  : English (ICLR 2026 submission text)
%%   ko  : Korean mirror (prose translated 1:1; equations, labels, cite keys,
%%         tables and figure paths unchanged)
%% ---------------------------------------------------------------------------
\newcommand{\paperLang}{en}  % en | ko

\usepackage{hyperref}
\usepackage{url}
\usepackage{booktabs}
\usepackage{graphicx}
\usepackage{array}
\usepackage{multirow}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{xcolor}
\usepackage{tikz}
\usepackage{listings}
\lstset{basicstyle=\ttfamily\footnotesize, breaklines=true, columns=fullflexible,
        breakindent=0pt, frame=single, framesep=4pt, xleftmargin=2pt, xrightmargin=2pt}

%% Korean glyph rendering --- only loaded when \paperLang is "ko".
%% kotex requires XeLaTeX (or LuaLaTeX); mirrors the ko branch of
%% main_kdduc_acm.tex. Korean fonts ship without italic shapes, so the
%% "Font shape ... undefined" substitution warnings are cosmetic.
\usepackage{ifthen}
\ifthenelse{\equal{\paperLang}{ko}}{\usepackage{kotex}}{}

%% Shared figure path --- both languages reuse the same images.
\graphicspath{{figures/}}

%% Draft markers. \todo prints in red so that every unfilled number, missing
%% citation and unverified claim is visible in the PDF. Remove before
%% camera-ready (grep for \todo).
\newcommand{\todo}[1]{\textcolor{red}{[TODO: #1]}}

%% Accessibility alt-text hook. The ACM class defines \Description for
%% screen-reader text on figures; this article class does not, so the v1
%% appendix figure (ported from the ACM build) needs a local stand-in. Kept
%% as a silent no-op here since ICLR has no equivalent rendering path for it.
\newcommand{\Description}[1]{}

%% Notation macros --- one definition, used in body and appendix alike.
\newcommand{\Xstar}{X^{\ast}}                 % score-equivalent of erasure (points)
\newcommand{\Fthreat}{F_{\mathrm{T}}}         % session forfeit rate, threat arm (exit cell)
\newcommand{\Fruler}{F}                       % ruler curve F(X)
\newcommand{\riconf}{\texttt{ri\_confidence}} % thinking tokens, confidence call
\newcommand{\ridec}{\texttt{ri\_forfeit}}     % thinking tokens, decision call
\newcommand{\ritask}{\texttt{ri\_task}}       % thinking tokens, task call
\newcommand{\pthreat}{\texttt{P\_THREAT}}     % self-reported event probability field

%% Line-breaking safety net for long typewriter identifiers.
\setlength{\emergencystretch}{3em}
\tolerance=2000
\hbadness=10000
