%% LLM Squid Game --- Nature Machine Intelligence (Article) entry point.
%% Class: natmi/style/sn-jnl.cls and natmi/style/sn-nature.bst are the OFFICIAL
%% files, copied unmodified from the Springer Nature LaTeX authoring template,
%% "journal article template package (December 2024 version)", template
%% Version 3.1, downloaded 2026-10-17 from
%%   https://www.springernature.com/gp/authors/campaigns/latex-author-support
%% which Nature Machine Intelligence's "AIP and formatting" page links as the
%% place to "download the Springer Nature LaTeX template". The full kit (class,
%% all .bst files, sample sn-article.tex/.pdf, user manual) is kept untouched
%% under natmi/official/sn-article-template/.
%%
%% Class options: [pdflatex,sn-nature] is the line the official sample marks
%% "Style for submissions to Nature Portfolio journals" (numbered references,
%% sn-nature.bst through natbib). The user manual also asks for [referee]
%% (double spacing) and allows [lineno] for review copies; uncomment below.
%% Fonts and layout are the class's own (Computer Modern, single column).
%%
%% Compile (both languages):  bash build_natmi.sh
%% Language is fixed by the wrapper files en_natmi.tex / ko_natmi.tex through
%% \paperLang (en | ko). English builds with pdfLaTeX, as the template does; the
%% Korean mirror needs XeLaTeX (kotex). Bibliography: BibTeX with sn-nature.bst,
%% as in the template, reading the shared references.bib.
%%
%% Final (post-acceptance) submission: the journal wants one .tex file with the
%% references pasted in from the .bbl (no \bibliography), so the \input tree
%% below must be flattened at that stage (e.g. latexpand).
%\documentclass[pdflatex,sn-nature,referee,lineno]{sn-jnl}% review copy
\documentclass[pdflatex,sn-nature]{sn-jnl}% Style for submissions to Nature Portfolio journals

%% ---------------------------------------------------------------------------
%% Language switch
%%   en  : English (Nature Machine Intelligence submission)
%%   ko  : Korean mirror (prose translated 1:1; equations, labels, cite keys,
%%         tables and figure paths unchanged)
%% ---------------------------------------------------------------------------
\providecommand{\paperLang}{en}  % en | ko

%% Standard packages, a subset of the ones the official sample loads.
\usepackage{graphicx}
\usepackage{amsmath,amssymb,amsfonts}
\usepackage{xcolor}
\usepackage{textcomp}
\usepackage{booktabs}

%% Korean glyph rendering --- only loaded when \paperLang is "ko".
\usepackage{ifthen}
%% sn-jnl.cls loads no text-font package, so under XeLaTeX kotex finds its
%% default Hangul faces itself. The fonts are still named explicitly, with the
%% same UnBatang/UnDotum (.ttf, shipped with TeX Live) as the CogSci and ICML
%% Korean builds, so the three Korean PDFs share one face.
\ifthenelse{\equal{\paperLang}{ko}}{%
  \usepackage{kotex}%
  \setmainhangulfont{UnBatang}[Extension=.ttf,BoldFont=UnBatangBold]%
  \setsanshangulfont{UnDotum}[Extension=.ttf,BoldFont=UnDotumBold]%
  \setmonohangulfont{UnDotum}[Extension=.ttf,BoldFont=UnDotumBold]%
}{}

%% Shared figure path --- both languages reuse the same images.
\graphicspath{{figures/}}

%% Draft markers, as in the ICML and CogSci builds. Remove before submission (grep \todo).
\newcommand{\todo}[1]{\textcolor{red}{[TODO: #1]}}

%% Empty-bibliography guard. BibTeX stops with "I found no \citation commands"
%% (an error, though latexmk carries on) while the text cites nothing. So
%% \bibliography only writes \bibdata and reads the .bbl once the previous
%% run's .aux holds a \citation; the first \cite therefore costs one extra
%% pass, which latexmk does on its own. No \nocite, no change to sn-jnl.cls.
\makeatletter
\newif\ifnatmi@cited
\def\citation#1{\global\natmi@citedtrue}% read only while the old .aux is loaded
\let\natmi@bibliography\bibliography
\renewcommand\bibliography[1]{\ifnatmi@cited\natmi@bibliography{#1}\fi}
\makeatother

%% Tables, vector figures (pgfplots) and the shared colour palette (same file
%% as cogsci2027/macros.tex). Packages and macros only.
\input{natmi/macros}

\begin{document}

%% Front matter (title, authors, abstract, keywords). sn-jnl.cls expects these
%% after \begin{document} and before \maketitle, as in the official sample.
\input{natmi/\paperLang/meta}

\maketitle

\input{natmi/\paperLang/content}

\end{document}
